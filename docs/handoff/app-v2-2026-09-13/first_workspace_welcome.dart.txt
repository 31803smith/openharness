import 'package:flutter/material.dart';

import '../shared/theme/app_theme.dart' as grid;
import '../state/app_state.dart';
import '../state/swarm_catalog.dart';
import 'engine_identity.dart';
import 'swarm_icon.dart';
import 'welcome_workspace_preview.dart';

/// Start with the work already available. Machine and project management is
/// available on demand, without making it a prerequisite for the first agent.
class FirstWorkspaceWelcome extends StatefulWidget {
  const FirstWorkspaceWelcome({
    super.key,
    required this.notifier,
    required this.groups,
    required this.searchField,
    required this.onNewAgent,
    required this.onLinkMachine,
    required this.onProject,
    required this.catalogBuilder,
    this.onChooseFolder,
    this.onCloneRepository,
    this.onAgent,
  });

  final AppNotifier notifier;
  final List<SwarmProjectGroup> groups;
  final Widget searchField;
  final VoidCallback onNewAgent, onLinkMachine;
  final VoidCallback? onChooseFolder, onCloneRepository;
  final ValueChanged<SwarmAgentRef>? onAgent;
  final ValueChanged<SwarmProjectGroup> onProject;
  final WidgetBuilder catalogBuilder;

  @override
  State<FirstWorkspaceWelcome> createState() => _FirstWorkspaceWelcomeState();
}

class _FirstWorkspaceWelcomeState extends State<FirstWorkspaceWelcome> {
  bool _catalogOpen = false;

  @override
  Widget build(BuildContext context) {
    grid.AppTheme.watch(context);
    final app = widget.notifier;
    final ready =
        swarmAgents(app)
            .where(
              (entry) =>
                  !entry.machine.needsLink &&
                  entry.machine.nodeOnline != false &&
                  entry.agent.terminalAvailable,
            )
            .toList()
          ..sort((a, b) {
            if (a.machine.isLocalMachine != b.machine.isLocalMachine) {
              return a.machine.isLocalMachine ? -1 : 1;
            }
            return a.agent.name.toLowerCase().compareTo(
              b.agent.name.toLowerCase(),
            );
          });
    final local = app.machineStates.values
        .where(
          (machine) =>
              machine.isLocalMachine &&
              !machine.needsLink &&
              machine.nodeOnline != false,
        )
        .firstOrNull;
    final canCreate = app.machineStates.values.any(
      (machine) => !machine.needsLink && machine.nodeOnline != false,
    );
    final finding =
        app.machinesLoading ||
        app.machineStates.values.any(
          (machine) =>
              !machine.needsLink &&
              machine.nodeOnline != false &&
              (machine.agentLoadStatus == AgentLoadStatus.idle ||
                  machine.agentLoadStatus == AgentLoadStatus.loading),
        );
    final chooseFolder = local != null && widget.onChooseFolder != null;
    final existing = ready.isNotEmpty && widget.onAgent != null;
    final saved = widget.groups.where((group) => group.saved != null).toList();

    Widget primary() => FilledButton.icon(
      key: const ValueKey('first-workspace-start'),
      autofocus: !existing,
      onPressed: chooseFolder
          ? widget.onChooseFolder
          : canCreate
          ? widget.onNewAgent
          : finding
          ? null
          : widget.onLinkMachine,
      icon: Icon(
        chooseFolder
            ? Icons.folder_open_outlined
            : canCreate
            ? Icons.add
            : Icons.computer_outlined,
        size: 18,
      ),
      label: Text(
        chooseFolder
            ? 'Choose folder…'
            : canCreate
            ? 'New agent'
            : finding
            ? 'Finding your computers…'
            : 'Link a machine',
      ),
      style: FilledButton.styleFrom(
        minimumSize: const Size(168, 44),
        backgroundColor: grid.AppPalette.swarmAccent,
        foregroundColor: grid.AppPalette.swarmTabBar,
      ),
    );

    Widget intro() => Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Text(
          existing ? 'Your agents, ready to go' : 'Your agents, together',
          style: const TextStyle(
            fontSize: 32,
            height: 1.15,
            fontWeight: FontWeight.w600,
            letterSpacing: -0.7,
            color: Colors.white,
          ),
        ),
        const SizedBox(height: 12),
        Text(
          existing
              ? 'Bring your existing work into Harness. Choose an agent to continue.'
              : 'Work with Claude Code, Codex, and other agents side by side. Start with one in a folder you already use.',
          style: const TextStyle(
            fontSize: 14,
            height: 1.55,
            color: Colors.white70,
          ),
        ),
        if (!existing) ...[
          const SizedBox(height: 24),
          Wrap(
            spacing: 12,
            runSpacing: 8,
            crossAxisAlignment: WrapCrossAlignment.center,
            children: [
              primary(),
              if (local != null && widget.onCloneRepository != null)
                TextButton(
                  key: const ValueKey('first-workspace-clone'),
                  onPressed: widget.onCloneRepository,
                  style: TextButton.styleFrom(foregroundColor: Colors.white70),
                  child: const Text('Clone repository…'),
                ),
            ],
          ),
          const SizedBox(height: 12),
          const Text(
            'Choose a folder → Pick an agent → Give it a task',
            style: TextStyle(fontSize: 12, height: 1.5, color: Colors.white60),
          ),
        ],
      ],
    );

    const example = Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Row(
          children: [
            Text(
              'Work side by side',
              style: TextStyle(
                fontSize: 12,
                fontWeight: FontWeight.w600,
                color: Colors.white70,
              ),
            ),
            Spacer(),
            Text(
              'Example',
              style: TextStyle(fontSize: 11, color: Colors.white54),
            ),
          ],
        ),
        SizedBox(height: 10),
        WelcomeWorkspacePreview(),
        SizedBox(height: 10),
        Text(
          'A swarm is a tab of agents. Add another with + when you’re ready.',
          style: TextStyle(fontSize: 12, height: 1.5, color: Colors.white60),
        ),
      ],
    );

    return LayoutBuilder(
      builder: (context, constraints) => SingleChildScrollView(
        child: ConstrainedBox(
          constraints: BoxConstraints(minHeight: constraints.maxHeight),
          child: Center(
            child: Container(
              constraints: const BoxConstraints(maxWidth: 1040),
              padding: EdgeInsets.symmetric(
                horizontal: constraints.maxWidth < 700 ? 24 : 48,
                vertical: 40,
              ),
              child: Column(
                mainAxisSize: MainAxisSize.min,
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  if (existing) ...[
                    intro(),
                    const SizedBox(height: 24),
                    widget.searchField,
                    const SizedBox(height: 16),
                    _Surface(
                      child: Column(
                        children: [
                          for (final entry in ready.take(3))
                            ListTile(
                              key: ValueKey(
                                'welcome-agent:${entry.machineId}:${entry.agent.id}',
                              ),
                              leading: EngineMark(
                                engine: entry.agent.engine,
                                size: 22,
                              ),
                              title: Text(
                                entry.agent.name,
                                maxLines: 1,
                                overflow: TextOverflow.ellipsis,
                                style: const TextStyle(
                                  fontSize: 14,
                                  color: Colors.white,
                                ),
                              ),
                              subtitle: Text(
                                [
                                  entry.machine.machine.displayName,
                                  entry.project?.name,
                                ].whereType<String>().join(' · '),
                                maxLines: 1,
                                overflow: TextOverflow.ellipsis,
                                style: const TextStyle(
                                  fontSize: 12,
                                  color: Colors.white60,
                                ),
                              ),
                              trailing: const Icon(
                                Icons.arrow_forward,
                                size: 16,
                                color: Colors.white54,
                              ),
                              onTap: () => widget.onAgent!(entry),
                            ),
                        ],
                      ),
                    ),
                    const SizedBox(height: 12),
                    TextButton.icon(
                      onPressed: widget.onNewAgent,
                      icon: const Icon(Icons.add, size: 17),
                      style: TextButton.styleFrom(
                        foregroundColor: Colors.white70,
                      ),
                      label: const Text('New agent'),
                    ),
                  ] else ...[
                    LayoutBuilder(
                      builder: (context, size) => size.maxWidth >= 720
                          ? Row(
                              crossAxisAlignment: CrossAxisAlignment.center,
                              children: [
                                Expanded(flex: 6, child: intro()),
                                const SizedBox(width: 40),
                                const Expanded(flex: 5, child: example),
                              ],
                            )
                          : Column(
                              crossAxisAlignment: CrossAxisAlignment.start,
                              children: [
                                intro(),
                                const SizedBox(height: 28),
                                example,
                              ],
                            ),
                    ),
                    if (finding) ...[
                      const SizedBox(height: 24),
                      const Semantics(
                        liveRegion: true,
                        child: Text(
                          'Looking for existing agents…',
                          style: TextStyle(fontSize: 12, color: Colors.white60),
                        ),
                      ),
                    ],
                  ],
                  if (saved.isNotEmpty && !_catalogOpen) ...[
                    const SizedBox(height: 28),
                    const Text(
                      'Your folders',
                      style: TextStyle(
                        fontSize: 13,
                        fontWeight: FontWeight.w600,
                        color: Colors.white70,
                      ),
                    ),
                    const SizedBox(height: 10),
                    _Surface(
                      child: Column(
                        children: [
                          for (final group in saved.take(3))
                            ListTile(
                              key: ValueKey(
                                'first-workspace-project:${group.id}',
                              ),
                              leading: const SwarmIcon(color: Colors.white60),
                              title: Text(
                                group.name,
                                maxLines: 1,
                                overflow: TextOverflow.ellipsis,
                                style: const TextStyle(
                                  fontSize: 13,
                                  color: Colors.white,
                                ),
                              ),
                              subtitle: Text(
                                group.saved!.path,
                                maxLines: 1,
                                overflow: TextOverflow.ellipsis,
                                style: const TextStyle(
                                  fontSize: 12,
                                  color: Colors.white60,
                                ),
                              ),
                              onTap: () => widget.onProject(group),
                            ),
                        ],
                      ),
                    ),
                  ],
                  const SizedBox(height: 24),
                  TextButton.icon(
                    key: const ValueKey('first-workspace-catalog'),
                    onPressed: () =>
                        setState(() => _catalogOpen = !_catalogOpen),
                    icon: Icon(
                      _catalogOpen ? Icons.expand_less : Icons.expand_more,
                      size: 18,
                    ),
                    label: const Text('Browse machines and projects'),
                    style: TextButton.styleFrom(
                      foregroundColor: Colors.white60,
                    ),
                  ),
                  if (_catalogOpen) ...[
                    const SizedBox(height: 12),
                    widget.catalogBuilder(context),
                  ],
                ],
              ),
            ),
          ),
        ),
      ),
    );
  }
}

class _Surface extends StatelessWidget {
  const _Surface({required this.child});
  final Widget child;
  @override
  Widget build(BuildContext context) => Container(
    padding: const EdgeInsets.all(6),
    decoration: BoxDecoration(
      color: grid.AppPalette.swarmSearchSurface,
      borderRadius: BorderRadius.circular(12),
      border: Border.all(color: Colors.white12),
    ),
    child: child,
  );
}
